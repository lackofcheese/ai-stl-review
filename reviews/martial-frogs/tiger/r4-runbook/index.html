How the round is judged

All checks are FAST and local:

  • prove the input SHA matches the Round 3x lineage record and the frozen R4 manifest;
  • apply only one uniform affine transform: 30.0 mm Z height, XY bounds centered, minimum Z at 0.0 mm; independently prove triangle count, watertightness, winding and coordinates were preserved by that transform;
  • require one component, zero open edges, zero non-manifold edges, zero degenerate faces and zero duplicate faces at final scale;
  • record exact dimensions, volume and high-pass surface roughness in the new lineage artifact's clean block;
  • run the tight 0.28 mm thin-connection probe and inspect every reported neck; run the 0.10 mm handle spectrum because the raw shell's Euler number is -2, require its enclosed-cavity count to remain zero, and inspect recorded mesh.section cross-sections at seven evenly spaced interior heights for nested contours, hidden shells or intersecting geometry; any unexplained internal contour stops the round before provisional review;
  • use eight-angle and close renders to distinguish intentional webbing or limb gaps from reconstruction noise and to reconcile any intentional section contour with visible exterior geometry;
  • inspect both sole patches against the Z=0 plane and confirm the occupied footprint fits a separate 32 mm round base; no base is fused in this round;
  • have a fresh-context reviewer compare the final eight-angle render, hands, feet, toggles and rear garment against the selected R3x mesh. Since placement is affine-only, any visual change is a hard failure.

The mechanical report and fresh-context audit produce a PROVISIONAL private R4 review with a clearly labelled decimated display proxy in its interactive viewer. The exact STL is handed over separately through Drive. Dimitri and Jawaknight's live-3D call on that exact file is the final print gate.

SECTION_DIR="$P/generated/r4-cross-sections/tiger-r3x-min-gptimage2-m1-clean-30mm"
SECTION_REPORT="$P/notes/r4-min-gptimage2-m1-internal-void.json"

python3 - "$CLEAN" "$SECTION_DIR" "$SECTION_REPORT" <<'PY'
import json
import pathlib
import sys

import numpy as np
import trimesh
from matplotlib.path import Path as MplPath

mesh_path, section_dir, report_path = map(pathlib.Path, sys.argv[1:])
mesh = trimesh.load_mesh(mesh_path, process=False)
if not isinstance(mesh, trimesh.Trimesh):
    raise TypeError(f"expected one Trimesh, got {type(mesh).__name__}")

section_dir.mkdir(parents=True, exist_ok=True)
z_min, z_max = map(float, mesh.bounds[:, 2])
fractions = np.linspace(0.125, 0.875, 7)
rows = []
for index, fraction in enumerate(fractions, 1):
    z_mm = z_min + float(fraction) * (z_max - z_min)
    section = mesh.section(
        plane_origin=[0.0, 0.0, z_mm], plane_normal=[0.0, 0.0, 1.0]
    )
    if section is None:
        raise RuntimeError(f"no cross-section at z={z_mm:.6f} mm")
    planar, _ = section.to_2D()
    loops = [np.asarray(loop)[:, :2] for loop in planar.discrete]
    closed = [
        len(loop) >= 4 and np.allclose(loop[0], loop[-1], atol=1e-7)
        for loop in loops
    ]
    if not loops or not all(closed):
        raise RuntimeError(f"open or empty cross-section at z={z_mm:.6f} mm")

    nested_pairs = []
    for inner_index, inner in enumerate(loops):
        for outer_index, outer in enumerate(loops):
            if inner_index == outer_index:
                continue
            inside = MplPath(outer).contains_points(inner[:-1], radius=-1e-9)
            if len(inside) and bool(inside.all()):
                nested_pairs.append([inner_index, outer_index])

    svg_name = f"section-{index:02d}.svg"
    planar.export(file_obj=section_dir / svg_name, file_type="svg")
    rows.append({
        "index": index,
        "fraction_of_height": round(float(fraction), 6),
        "z_mm": round(z_mm, 6),
        "closed_contours": len(loops),
        "all_contours_closed": True,
        "nested_contour_pairs": nested_pairs,
        "svg": str(section_dir / svg_name),
    })

report = {
    "schema_version": 1,
    "stl": str(mesh_path),
    "method": "trimesh.mesh.section at seven interior Z heights",
    "sections": rows,
    "nested_contour_pairs_total": sum(
        len(row["nested_contour_pairs"]) for row in rows
    ),
    "review_status": "pending-human-section-inspection",
}
report_path.write_text(json.dumps(report, indent=2) + "\n")
print(json.dumps(report, indent=2))
PY
